# The read log (read_log.h) is a library so tests/ can check it.
add_library(dsperate_headless_read_log STATIC read_log.cpp)
target_link_libraries(dsperate_headless_read_log PUBLIC dsperate_core)
add_executable(dsperate-headless main.cpp)
target_link_libraries(dsperate-headless PRIVATE dsperate_headless_read_log)
target_link_libraries(dsperate-headless PRIVATE dsperate_core)
if(DSPERATE_NET)
  target_link_libraries(dsperate-headless PRIVATE dsperate_net)
  target_compile_definitions(dsperate-headless PRIVATE DSPERATE_NET=1)
endif()
if(DSPERATE_CHEEVOS)
  target_link_libraries(dsperate-headless PRIVATE dsperate_cheevos)
  target_compile_definitions(dsperate-headless PRIVATE DSPERATE_CHEEVOS=1)
endif()
if(DSPERATE_LUA)
  find_package(Lua 5.4 QUIET)
  if(NOT LUA_FOUND OR LUA_VERSION_STRING VERSION_LESS 5.4)
    message(STATUS "DSperate: Lua 5.4 not found; dsperate-headless is built without --lua")
  endif()
endif()
if(DSPERATE_LUA AND LUA_FOUND AND NOT LUA_VERSION_STRING VERSION_LESS 5.4)
  # The --lua scripting host, a library so tests/ can drive it without a ROM.
  add_library(dsperate_headless_lua STATIC lua_host.cpp)
  target_link_libraries(dsperate_headless_lua PUBLIC dsperate_core ${LUA_LIBRARIES})
  target_include_directories(dsperate_headless_lua PRIVATE ${LUA_INCLUDE_DIR})
  target_link_libraries(dsperate-headless PRIVATE dsperate_headless_lua)
  target_compile_definitions(dsperate-headless PRIVATE DSPERATE_LUA=1)
endif()
